function waLink(phone, text) {
  let digits = String(phone || '').replace(/\D/g, '');
  if (digits.length === 8) digits = '223' + digits;
  return `https://wa.me/${digits}?text=${encodeURIComponent(text)}`;
}

function stmtDate(iso) {
  if (!iso) return '';
  try {
    return new Date(iso).toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: '2-digit' });
  } catch (e) {
    return '';
  }
}

const fmt = (n) => {
 const v = Math.round((n + Number.EPSILON) * 100) / 100;
 return v.toLocaleString('fr-FR', { maximumFractionDigits: 2 });
};

const fmtDate = (iso) => {
 if (!iso) return'';

};

const fmtDateTime = (iso) => {
 if (!iso) return'';
 return new Date(iso).toLocaleString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit' });

function downloadFile(filename, content, mime) {
 const blob = content instanceof Blob ? content : new Blob([content], { type: mime });
 const url = URL.createObjectURL(blob);
 const a = document.createElement('a');
 a.href = url;
 a.download = filename;
 document.body.appendChild(a);
 a.click();
 document.body.removeChild(a);
 URL.revokeObjectURL(url);
}

function rowsToCSVText(rows) {
 return rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g,'""')}"`).join(';')).join('\n');
}

function buildSummaryRows(clients) {
 const rows = [['Code','Detail','Total du','Total paye','Reste','Statut']];
 clients.forEach((c) => {
 const t = clientTotals(c);
 const statusLabel = { paye: 'Paye', partiel: 'Partiel', impaye: 'Impaye' }[t.status];
 rows.push([c.code, breakdownLine(c), t.totalDue, t.totalPaid, t.remaining, statusLabel]);
 });
 return rows;
}

function toCSV(clients) {
 return rowsToCSVText(buildSummaryRows(clients));
}

function buildDetailedRows(clients) {
 const rows = [['Code','Type','Poids (kg)','Tarif','Montant','Date','Statut client','Total du client','Total paye client','Reste client']];
 clients.forEach((c) => {
 const t = clientTotals(c);
 const statusLabel = { paye: 'Paye', partiel: 'Partiel', impaye: 'Impaye' }[t.status];
 if (c.carryOver > 0) {
 rows.push([c.code,'Report semaine precedente','','', c.carryOver,'', statusLabel, t.totalDue, t.totalPaid, t.remaining]);
 }
 c.loads.forEach((l) => {
 rows.push([c.code, l.truck ? `Chargement ${l.truck}` : 'Chargement', l.weight, l.rate, l.weight * l.rate, fmtDate(l.date), statusLabel, t.totalDue, t.totalPaid, t.remaining]);
 });
 c.payments.forEach((p) => {
 rows.push([c.code, p.method && PAYMENT_METHOD_LABELS[p.method] ? `Paiement ${PAYMENT_METHOD_LABELS[p.method]}` : 'Paiement', '', '', p.amount, fmtDate(p.date), statusLabel, t.totalDue, t.totalPaid, t.remaining]);
 });
 if (c.loads.length === 0 && c.payments.length === 0 && !c.carryOver) {
 rows.push([c.code,'Aucune donnee','','','','', statusLabel, t.totalDue, t.totalPaid, t.remaining]);
 }
 });
 return rows;
}

function toDetailedCSV(clients) {
 return rowsToCSVText(buildDetailedRows(clients));
}

const uid = () => Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4);

function parseFrenchNumber(str) {
 if (str === undefined || str === null) return NaN;
 const cleaned = String(str).replace(/[\s\u00A0\u202F]/g,'').replace(',','.');
 return parseFloat(cleaned);
}

function parseCSVText(text) {
 const clean = text.replace(/^\uFEFF/,'');
 const delimiter = (clean.split('\n')[0].match(/;/g) || []).length >= (clean.split('\n')[0].match(/,/g) || []).length ? ';' : ',';
 const lines = clean.split(/\r?\n/).filter((l) => l.trim().length > 0);
 return lines.map((line) => {
 const cells = [];
 let cur ='';
 let inQuotes = false;
 for (let i = 0; i < line.length; i++) {
 const ch = line[i];
 if (inQuotes) {
 if (ch ==='"') {
 if (line[i + 1] ==='"') { cur += '"'; i++; }
 else inQuotes = false;
 } else cur += ch;
 } else {
 if (ch ==='"') inQuotes = true;
 else if (ch === delimiter) { cells.push(cur); cur =''; }
 else cur += ch;
 }
 }
 cells.push(cur);
 return cells;
 });
}

function parseDetailToLoads(detail) {
 if (!detail || detail ==='—') return [];
 const loads = [];
 const re = /([\d\s\u00A0\u202F.,]+)\s*kg\s*[×x*]\s*([\d\s\u00A0\u202F.,]+)/gi;
 let m;
 while ((m = re.exec(detail)) !== null) {
 const weight = parseFrenchNumber(m[1]);
 const rate = parseFrenchNumber(m[2]);
 if (!isNaN(weight) && !isNaN(rate)) {
 loads.push({ id: uid(), weight, rate, date: new Date().toISOString() });
 }
 }
 return loads;
}

function startOfDay(d) {
 return new Date(d.getFullYear(), d.getMonth(), d.getDate());
}

function isDateInPeriod(iso, period, customStart, customEnd) {
 if (!iso) return false;
 const d = new Date(iso);
 const now = new Date();
 if (period === 'today') {
 return d >= startOfDay(now);
 }
 if (period === 'yesterday') {
 const y = new Date(now);
 y.setDate(y.getDate() - 1);
 return d >= startOfDay(y) && d < startOfDay(now);
 }
 if (period === 'week') {
 const s = new Date(now);
 s.setDate(s.getDate() - 6);
 return d >= startOfDay(s);
 }
 if (period === 'custom') {
 if (!customStart || !customEnd) return false;
 const s = new Date(customStart +'T00:00:00');
 const e = new Date(customEnd +'T23:59:59');
 return d >= s && d <= e;
 }
 return true; //'all'
}

function buildPaymentsPeriodRows(clients, period, customStart, customEnd) {
 const rows = [['Client','Date','Heure','Montant','Mode']];
 let total = 0;
 const byMethod = {};
 clients.forEach((c) => {
 c.payments
 .filter((p) => isDateInPeriod(p.date, period, customStart, customEnd))
 .sort((a, b) => a.date.localeCompare(b.date))
 .forEach((p) => {
 const dt = new Date(p.date);
 rows.push([c.code, fmtDate(p.date), dt.toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' }), p.amount, PAYMENT_METHOD_LABELS[p.method] || '']);
 total += p.amount;
 const mk = PAYMENT_METHOD_LABELS[p.method] || 'Non précisé';
 byMethod[mk] = (byMethod[mk] || 0) + p.amount;
 });
 });
 rows.push(['','','TOTAL', total, '']);
 Object.keys(byMethod).forEach((k) => rows.push(['','', `Total ${k}`, byMethod[k], '']));
 return rows;
}

function toPaymentsPeriodCSV(clients, period, customStart, customEnd) {
 return rowsToCSVText(buildPaymentsPeriodRows(clients, period, customStart, customEnd));
}

function localDateStr(d) {
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

function addDaysStr(n) {
  const d = new Date();
  d.setDate(d.getDate() + n);
  return localDateStr(d);
}

function clientTotals(client) {
 const loadsTotal = client.loads.reduce((s, l) => s + l.weight * l.rate, 0);
 const carryOver = client.carryOver || 0;
 const totalDue = loadsTotal + carryOver;
 const totalPaid = client.payments.reduce((s, p) => s + p.amount, 0);
 const remaining = Math.max(0, totalDue - totalPaid);
 let status ='impaye';
 if (totalDue > 0 && totalPaid >= totalDue) status ='paye';
 else if (totalPaid > 0) status ='partiel';
 return { totalDue, totalPaid, remaining, status, carryOver };
}

function breakdownLine(client) {
 const parts = [];
 if (client.carryOver > 0) parts.push(`Report ${fmt(client.carryOver)}`);
 client.loads.forEach((l) => parts.push(`${fmt(l.weight)}kg × ${fmt(l.rate)}`));
 if (parts.length === 0) return'—';
 return parts.join(' + ');
}

function getLastKnownRate(client, archives) {
 if (client.loads.length > 0) {
 return client.loads[client.loads.length - 1].rate;
 }
 for (const a of archives) {
 const match = a.clients.find((c) => c.code === client.code);
 if (match && match.loads.length > 0) {
 return match.loads[match.loads.length - 1].rate;
 }
 }
 return null;
}

 // A failed read must never look like an empty app: the app retries, and otherwise shows an error
 // screen. Nothing is written before everything was read, a key is never overwritten with data this
 // session did not load, failed writes are shown and retried, and a safety copy is kept each day.

 const [saveProblem, setSaveProblem] = useState('');
 const [safetyCopies, setSafetyCopies] = useState([]);
 const loadedRef = useRef(false);
 const keyStatus = useRef({});
 const blockedKeys = useRef({});
 const lastSaved = useRef({});
 const saveFailures = useRef({});
 const saveTimers = useRef({});
 const latestData = useRef({});

 const KEY_MAP = { clients: STORAGE_KEY, settings: SETTINGS_KEY, archives: ARCHIVES_KEY, expenses: EXPENSES_KEY };
 const SAFETY_KEYS = ['recouvrement-safety-1', 'recouvrement-safety-2'];

 const storageGet = async (key) => {
 try {
 const r = await window.storage.get(key, false);
 if (r && typeof r.value === 'string') return { status: 'value', value: r.value };
 return { status: 'absent' };
 } catch (e) {
 return { status: 'error' };
 }
 };

 const storageKeys = async () => {
 try {
 if (!window.storage || typeof window.storage.list !== 'function') return null;
 const res = await window.storage.list('recouvrement-', false);
 const arr = Array.isArray(res) ? res : res && Array.isArray(res.keys) ? res.keys : null;
 if (!arr) return null;
 return new Set(arr.map((k) => (typeof k === 'string' ? k : k && (k.key || k.name))).filter(Boolean));
 } catch (e) {
 return null;
 }
 };

 const storageSet = async (key, value) => {
 const r = await window.storage.set(key, value, false);
 if (r === null) throw new Error('storage.set failed');
 };

 const refreshSafetyCopies = useCallback(async () => {
 const out = [];
 for (const key of SAFETY_KEYS) {
 const r = await storageGet(key);
 if (r.status !== 'value') continue;
 try {
 const d = JSON.parse(r.value);
 if (d && Array.isArray(d.clients)) out.push({ key, savedAt: d.savedAt, day: d.day, count: d.clients.length });
 } catch (e) {}
 }
 setSafetyCopies(out);
 return out;
 }, []);

 // Once a day, keep the data as it was when the app was opened (two days are kept).
 const writeSafetyCopy = async (data) => {
 if (!data.clients || data.clients.length === 0) return;
 const today = localDateStr(new Date());
 let cur = await storageGet(SAFETY_KEYS[0]);
 if (cur.status === 'error') {
 // A missing key also fails to read: only leave the slot alone if it exists but cannot be read now.
 const keys = await storageKeys();
 if (keys && keys.has(SAFETY_KEYS[0])) return;
 cur = { status: 'absent' };
 }
 let curDay = '';
 if (cur.status === 'value') {
 try {
 curDay = JSON.parse(cur.value).day || '';
 } catch (e) {}
 }
 if (curDay === today) return;
 try {
 if (cur.status === 'value') await storageSet(SAFETY_KEYS[1], cur.value);
 await storageSet(SAFETY_KEYS[0], JSON.stringify({ day: today, savedAt: new Date().toISOString(), clients: data.clients, settings: data.settings, expenses: data.expenses }));
 } catch (e) {}
 };

 const loadAll = useCallback(async (firstUse) => {
 setLoadError('');
 const names = ['clients', 'settings', 'archives', 'expenses'];
 let results = null;
 let statuses = null;
 let listAvailable = false;
 for (let attempt = 0; attempt < 4; attempt++) {
 if (attempt) await new Promise((r) => setTimeout(r, 800 * attempt));
 results = await Promise.all(names.map((n) => storageGet(KEY_MAP[n])));
 statuses = results.map((r) => r.status);
 if (!statuses.includes('error')) break;
 // A missing key also makes get() fail: the key list tells the two cases apart.
 const keys = await storageKeys();
 if (keys) {
 listAvailable = true;
 statuses = statuses.map((s, i) => (s === 'error' && !keys.has(KEY_MAP[names[i]]) ? 'absent' : s));
 if (!statuses.includes('error')) break;
 }
 }
 if (statuses.includes('error')) {
 // Small keys that keep failing while the others answer are taken as not created yet; they are
 // checked again before their first write. The clients and the history are never guessed.
 const answered = statuses.some((s) => s === 'value');
 statuses = statuses.map((s, i) => (s === 'error' && answered && (names[i] === 'settings' || names[i] === 'expenses') ? 'assumed' : s));
 }
 if (statuses.includes('error') && !firstUse) {
 setLoadError(!listAvailable && !statuses.includes('value') ? 'nothing' : 'failed');
 return;
 }
 const parsed = {};
 for (let i = 0; i < names.length; i++) {
 const n = names[i];
 if (statuses[i] !== 'value') continue;
 try {
 parsed[n] = JSON.parse(results[i].value);
 } catch (e) {
 setLoadError('corrupt');
 return;
 }
 }
 names.forEach((n, i) => {
 const s = statuses[i];
 keyStatus.current[n] = s === 'value' ? 'loaded' : s === 'error' ? 'assumed' : s;
 });
 // Remember exactly what is stored, so opening the app never rewrites anything.
 const lists = { clients: setClients, archives: setArchives, expenses: setExpenses };
 Object.keys(lists).forEach((n) => {
 const value = Array.isArray(parsed[n]) ? parsed[n] : [];
 lastSaved.current[n] = JSON.stringify(value);
 lists[n](value);
 });
 const storedSettings = parsed.settings && typeof parsed.settings === 'object' ? parsed.settings : null;
 setSettings((s) => {
 const merged = storedSettings ? { ...s, ...storedSettings } : s;
 lastSaved.current.settings = JSON.stringify(merged);
 return merged;
 });
 if (storedSettings && storedSettings.pin) setUnlocked(false);
 loadedRef.current = true;
 setLoaded(true);
 writeSafetyCopy({ clients: parsed.clients || [], settings: parsed.settings || null, expenses: parsed.expenses || [] }).then(refreshSafetyCopies);
 }, [refreshSafetyCopies]);

 useEffect(() => {

 }, [loadAll]);

 const flushSave = useCallback(async (which) => {
 if (saveTimers.current[which]) {
 clearTimeout(saveTimers.current[which]);
 saveTimers.current[which] = null;
 }
 if (!loadedRef.current || blockedKeys.current[which]) return;
 const json = JSON.stringify(latestData.current[which]);
 if (json === lastSaved.current[which]) return;
 const key = KEY_MAP[which];
 if (keyStatus.current[which] === 'assumed') {
 const check = await storageGet(key);
 if (check.status === 'value') {
 blockedKeys.current[which] = true;
 setSaveProblem("Des données enregistrées n'ont pas pu être lues à l'ouverture. Ferme et rouvre l'app avant de continuer : rien n'a été écrasé.");
 return;
 }
 keyStatus.current[which] = 'absent';
 }
 try {
 await storageSet(key, json);
 lastSaved.current[which] = json;
 saveFailures.current[which] = 0;
 if (!Object.values(saveFailures.current).some((n) => n > 0) && !Object.values(blockedKeys.current).some(Boolean)) setSaveProblem('');
 } catch (e) {
 const n = (saveFailures.current[which] || 0) + 1;
 saveFailures.current[which] = n;
 setSaveProblem("Enregistrement impossible pour l'instant (connexion ?). Tes dernières saisies restent affichées et seront enregistrées dès que possible : ne ferme pas l'app.");
 saveTimers.current[which] = setTimeout(() => flushSave(which), Math.min(30000, 3000 * n));
 }
 }, []);

 return;

 const codeIdx = header.findIndex((h) => h ==='code');
 const detailIdx = header.findIndex((h) => h.startsWith('detail'));

 if (codeIdx === -1 || payeIdx === -1) {

 clients.forEach((c) => { clientByCode[c.code.toUpperCase()] = c; });

 const row = rows[i];
 const code = (row[codeIdx] ||'').trim().toUpperCase();
 if (!code) continue;
 const newPaye = parseFrenchNumber(row[payeIdx]);
 if (isNaN(newPaye)) continue;
 const detail = detailIdx >= 0 ? row[detailIdx] :'';
 const existing = clientByCode[code];
 if (existing) {
 const currentTotals = clientTotals(existing);
 const delta = Math.round((newPaye - currentTotals.totalPaid) * 100) / 100;

 updates.push({ clientId: existing.id, code, oldPaye: currentTotals.totalPaid, newPaye, delta });
 }

 const loads = parseDetailToLoads(detail);
 newClients.push({ code, loads, payments: newPaye > 0 ? [{ amount: newPaye }] :[] });
 }
 }
 setCsvImportError('');
 setPendingCsvImport({ updates, newClients });
 } catch (err) {
 setCsvImportError("Impossible de lire ce fichier CSV.");
 setPendingCsvImport(null);

 setShowCsvImportConfirm(true);

function SignaturePad({ value, onChange }) {
 const canvasRef = useRef(null);
 const drawingRef = useRef(false);
 const lastPoint = useRef(null);
 const hasInkRef = useRef(false);

 const getPos = (e) => {
 const rect = canvasRef.current.getBoundingClientRect();
 const t = e.touches && e.touches[0] ? e.touches[0] : e;
 return { x: t.clientX - rect.left, y: t.clientY - rect.top };
 };

 const start = (e) => {
 e.preventDefault();
 drawingRef.current = true;
 lastPoint.current = getPos(e);
 };

 const move = (e) => {
 if (!drawingRef.current) return;
 e.preventDefault();
 const canvas = canvasRef.current;
 const ctx = canvas.getContext('2d');
 const p = getPos(e);
 ctx.strokeStyle ='#111';
 ctx.lineWidth = 2.2;
 ctx.lineCap ='round';
 ctx.lineJoin ='round';
 ctx.beginPath();
 ctx.moveTo(lastPoint.current.x, lastPoint.current.y);
 ctx.lineTo(p.x, p.y);
 ctx.stroke();
 lastPoint.current = p;
 hasInkRef.current = true;
 };

 const end = () => {
 if (!drawingRef.current) return;
 drawingRef.current = false;
 if (hasInkRef.current && onChange) onChange(canvasRef.current.toDataURL('image/png'));
 };

 const clear = () => {
 const canvas = canvasRef.current;
 const ctx = canvas.getContext('2d');
 ctx.clearRect(0, 0, canvas.width, canvas.height);
 hasInkRef.current = false;
 if (onChange) onChange(null);
 };

 return (
 <div>
 <canvas
 ref={canvasRef}
 width={600}
 height={180}
 style={{ width: '100%', height: '130px', background: '#fff', borderRadius: '10px', touchAction: 'none', display: 'block' }}
 onMouseDown={start}
 onMouseMove={move}
 onMouseUp={end}
 onMouseLeave={end}
 onTouchStart={start}
 onTouchMove={move}
 onTouchEnd={end}
 />
 <div className="flex items-center justify-between mt-2">
 <div style={{ color: C.textFaint, fontSize: '11.5px' }}>Signe avec le doigt directement à l'écran</div>
 <button onClick={clear} className="flex items-center gap-1.5 px-3 py-1.5 rounded-2xl" style={{ background: C.surface2, border: `1px solid ${C.border}` }}>
 <Eraser size={13} color={C.textSecondary} />
 <span style={{ color: C.textSecondary, fontSize: '12px', fontFamily:"'Oswald', sans-serif", fontWeight: 600 }}>Effacer</span>
 </button>
 </div>
 </div>
 );
}

function roundRect(ctx, x, y, w, h, r) {
 ctx.beginPath();
 ctx.moveTo(x + r, y);
 ctx.arcTo(x + w, y, x + w, y + h, r);
 ctx.arcTo(x + w, y + h, x, y + h, r);
 ctx.arcTo(x, y + h, x, y, r);
 ctx.arcTo(x, y, x + w, y, r);
 ctx.closePath();
}

function expenseLabel(x) {

// "Ordre de saisie": by the date of the client's first load; clients without loads go last.
function saisieDate(c) {
  const loads = (c && c.loads) || [];
  return loads.length === 0 ? null : loads.reduce((min, l) => (l.date < min ? l.date : min), loads[0].date);
}

function compareSaisie(ca, cb) {
  const da = saisieDate(ca);
  const db = saisieDate(cb);
  if (da === null && db === null) return 0;
  if (da === null) return 1;
  if (db === null) return -1;
  return da.localeCompare(db);
}

function computeDayCash(day, clients, archives, expenses) {
  const onDay = (iso) => !!iso && localDateStr(new Date(iso)) === day;
  const activeById = new Map(clients.map((c) => [c.id, c]));
  const payments = [];
  const seen = new Set();
  const collect = (list) =>
    list.forEach((c) =>
      (c.payments || []).forEach((p) => {
        if (!onDay(p.date)) return;
        const key = p.id || `${c.id}-${p.date}-${p.amount}`;
        if (seen.has(key)) return;
        seen.add(key);
        payments.push({ clientId: c.id, code: c.code, amount: p.amount, method: p.method, date: p.date, client: c });
      })
    );
  collect(clients);
  archives.forEach((a) => collect(a.clients || []));
  payments.sort((a, b) => String(a.date).localeCompare(String(b.date)));
  const byClient = new Map();
  payments.forEach((p) => {
    const cur = byClient.get(p.clientId) || { clientId: p.clientId, code: p.code, amount: 0, client: p.client };
    cur.amount += p.amount;
    byClient.set(p.clientId, cur);
  });
  const order = new Map(clients.map((c, i) => [c.id, i]));
  const rows = [...byClient.values()]
    .sort((a, b) => (order.has(a.clientId) ? order.get(a.clientId) : 1e9) - (order.has(b.clientId) ? order.get(b.clientId) : 1e9))
    .sort((a, b) => compareSaisie(a.client, b.client))
    .map((r) => ({ ...r, inActive: activeById.has(r.clientId), reste: clientTotals(r.client).remaining }));
  const byMethod = {};
  payments.forEach((p) => {
    const k = p.method || 'inconnu';
    byMethod[k] = (byMethod[k] || 0) + p.amount;
  });
  const cashed = payments.reduce((s, p) => s + p.amount, 0);
  const dayExpenses = expenses.filter((x) => onDay(x.date)).sort((a, b) => String(a.createdAt || a.date).localeCompare(String(b.createdAt || b.date)));
  const spent = dayExpenses.reduce((s, x) => s + x.amount, 0);
  return { payments, rows, byMethod, cashed, dayExpenses, spent, rest: cashed - spent };
}

function drawCaisseCanvas(title, data, settings) {
  const W = 1080;
  const PAD = 56;
  const COL_PAYE = 860;
  const COL_RESTE = W - PAD;
  const LEFT_MAX = 600;
  const font = (size, weight) => `${weight || 400} ${size}px system-ui, -apple-system, 'Segoe UI', sans-serif`;
  const measure = document.createElement('canvas').getContext('2d');
  const wrap = (text, size, maxW) => {
    measure.font = font(size);
    const lines = [];
    let cur = '';
    String(text || '')
      .split(' ')
      .forEach((w) => {
        const t = cur ? `${cur} ${w}` : w;
        if (!cur || measure.measureText(t).width <= maxW) cur = t;
        else {
          lines.push(cur);
          cur = w;
        }
      });
    if (cur) lines.push(cur);
    return lines.slice(0, 4);
  };
  const items = [{ kind: 'thead', h: 58 }];
  if (!data.rows.length) items.push({ kind: 'empty', text: 'Aucun paiement', h: 56 });
  data.rows.forEach((r) => {
    const detail = wrap(breakdownLine(r.client), 23, LEFT_MAX);
    items.push({ kind: 'client', r, detail, h: 50 + detail.length * 32 + 12 });
  });
  items.push({ kind: 'total', label: 'Total encaissé', value: fmt(data.cashed), color: '#1e8449', h: 72 });
  items.push({ kind: 'section', label: 'Dépenses', h: 82 });
  if (!data.dayExpenses.length) items.push({ kind: 'empty', text: 'Aucune dépense', h: 56 });
  data.dayExpenses.forEach((x) => items.push({ kind: 'expense', label: expenseLabel(x), value: `− ${fmt(x.amount)}`, h: 54 }));
  items.push({ kind: 'total', label: 'Total dépenses', value: data.spent ? `− ${fmt(data.spent)}` : '0', color: '#c0392b', h: 72 });

      ctx.fillStyle = '#111';

      ctx.font = font(23);
      ctx.fillStyle = '#555';
      it.detail.forEach((ln, i) => ctx.fillText(ln, PAD, top + 42 + 32 * (i + 1)));

    } else if (it.kind === 'section') {
      ctx.textAlign = 'left';
      ctx.fillStyle = '#111';
      ctx.font = font(32, 700);
      ctx.fillText(it.label, PAD, top + 54);
      ctx.fillRect(PAD, top + it.h - 6, W - PAD * 2, 3);
    } else {
      ctx.textAlign = 'left';
      ctx.fillStyle = '#777';
      ctx.font = font(26);
      ctx.fillText(it.text, PAD, top + 38);
    }
    y += it.h;
  });
  const by = y + 44;
  ctx.strokeStyle = '#111';
  ctx.lineWidth = 4;
  stmtRoundRect(ctx, PAD, by, W - PAD * 2, 140, 18);
  ctx.stroke();
  ctx.textAlign = 'left';
  ctx.fillStyle = '#555';
  ctx.font = font(28);
  ctx.fillText('Reste en caisse', PAD + 28, by + 60);
  ctx.fillStyle = '#777';

  ctx.textAlign = 'right';
  ctx.fillStyle = data.rest >= 0 ? '#1e8449' : '#c0392b';
  ctx.font = font(56, 700);
  ctx.fillText(`${data.rest < 0 ? '− ' : ''}${fmt(Math.abs(data.rest))} F`, W - PAD - 28, by + 92);
  return canvas;
}

  const [sheet, setSheet] = useState(null);
  const [label, setLabel] = useState('');
  const [amount, setAmount] = useState('');
  const [confirmDelete, setConfirmDelete] = useState(false);
  const data = useMemo(() => computeDayCash(day, clients, archives, expenses), [day, clients, archives, expenses]);
  const dayDate = new Date(`${day}T12:00:00`);

    const d = new Date(`${day}T12:00:00`);
    d.setDate(d.getDate() + n);

function buildStatementWeeks(client, archives) {
  const weeks = [];
  const ordered = [...archives].sort((a, b) => String(a.archivedAt || '').localeCompare(String(b.archivedAt || '')));
  ordered.forEach((a) => {
    const c = (a.clients || []).find((x) => x.id === client.id);
    if (c && ((c.loads || []).length || (c.payments || []).length || c.carryOver)) weeks.push({ label: a.label || 'Semaine', closedAt: a.archivedAt, c });
  });
  weeks.push({ label: 'Semaine en cours', closedAt: null, c: client, current: true });
  let prevRemaining = null;
  return weeks.slice(-12).map((w) => {
    const t = clientTotals(w.c);
    const rows = [];
    if (w.c.carryOver > 0) {
      const penalty = prevRemaining !== null && w.c.carryOver - prevRemaining > 0.5 ? w.c.carryOver - prevRemaining : 0;
      rows.push({ date: null, text: penalty > 0 ? `Report semaine précédente (dont pénalité ${fmt(penalty)})` : 'Report semaine précédente', debit: w.c.carryOver });
    }
    const items = [];
    (w.c.loads || []).forEach((l) => items.push({ date: l.date, text: `Chargement ${fmt(l.weight)} kg × ${fmt(l.rate)}${l.truck ? ` (${l.truck})` : ''}`, debit: l.weight * l.rate }));
    (w.c.payments || []).forEach((p) =>
      items.push({ date: p.date, text: `Paiement${p.method && PAYMENT_METHOD_LABELS[p.method] ? ` ${PAYMENT_METHOD_LABELS[p.method]}` : ''}${p.label ? ` (${p.label === 'avance' ? 'avance' : 'reliquat'})` : ''}`, credit: p.amount })
    );
    items.sort((a, b) => String(a.date || '').localeCompare(String(b.date || '')));
    prevRemaining = t.remaining;
    return { label: w.label, closedAt: w.closedAt, current: !!w.current, rows: rows.concat(items), paid: t.totalPaid, remaining: t.remaining };
  });
}

function stmtRoundRect(ctx, x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.arcTo(x + w, y, x + w, y + h, r);
  ctx.arcTo(x + w, y + h, x, y + h, r);
  ctx.arcTo(x, y + h, x, y, r);
  ctx.arcTo(x, y, x + w, y, r);
  ctx.closePath();
}

function stmtTruncate(ctx, text, maxW) {
  if (ctx.measureText(text).width <= maxW) return text;
  let t = text;
  while (t.length > 1 && ctx.measureText(t + '…').width > maxW) t = t.slice(0, -1);
  return t + '…';
}

function drawStatementCanvas(client, allWeeks, settings) {
  const W = 1080;
  const PAD = 60;
  const LINE = 50;
  const HEADER = 330;
  const linesFor = (weeks) => {
    const lines = [];
    weeks.forEach((w) => {
      lines.push({ kind: 'week', left: w.label, right: w.current ? 'en cours' : w.closedAt ? `clôturée le ${stmtDate(w.closedAt)}` : '' });
      if (!w.rows.length) lines.push({ kind: 'row', left: 'Aucun mouvement', right: '', color: '#777' });
      w.rows.forEach((r) => lines.push({ kind: 'row', date: r.date ? stmtDate(r.date) : '', left: r.text, right: r.credit ? `− ${fmt(r.credit)}` : fmt(r.debit), color: r.credit ? '#1e8449' : '#111' }));
      lines.push({ kind: 'sub', left: w.current ? 'Reste à ce jour' : 'Reste de la semaine', right: fmt(w.remaining) });
    });
    return lines;
  };
  const heightOf = (lines) => HEADER + lines.reduce((s, l) => s + (l.kind === 'week' ? LINE + 26 : l.kind === 'sub' ? LINE + 10 : LINE), 0) + 120;
  let weeks = allWeeks;
  let lines = linesFor(weeks);
  while (heightOf(lines) > 15000 && weeks.length > 1) {
    weeks = weeks.slice(1);
    lines = linesFor(weeks);
  }
  const canvas = document.createElement('canvas');
  canvas.width = W;
  canvas.height = Math.min(heightOf(lines), 15000);
  const ctx = canvas.getContext('2d');
  const font = (size, weight) => `${weight || 400} ${size}px system-ui, -apple-system, 'Segoe UI', sans-serif`;
  ctx.fillStyle = '#ffffff';
  ctx.fillRect(0, 0, W, canvas.height);
  ctx.textAlign = 'left';
  ctx.fillStyle = '#111';
  ctx.font = font(44, 700);
  ctx.fillText(stmtTruncate(ctx, settings.businessName || 'Relevé de compte', 560), PAD, 90);
  if (settings.businessPhone) {
    ctx.font = font(28);
    ctx.fillStyle = '#555';
    ctx.fillText(settings.businessPhone, PAD, 134);
  }
  ctx.textAlign = 'right';
  ctx.fillStyle = '#111';
  ctx.font = font(32, 600);
  ctx.fillText('Relevé de compte', W - PAD, 90);
  ctx.font = font(26);
  ctx.fillStyle = '#555';
  ctx.fillText(stmtDate(new Date().toISOString()), W - PAD, 132);
  ctx.strokeStyle = '#111';
  ctx.lineWidth = 4;
  stmtRoundRect(ctx, PAD, 170, W - PAD * 2, 120, 18);
  ctx.stroke();
  ctx.textAlign = 'left';
  ctx.fillStyle = '#555';
  ctx.font = font(24);
  ctx.fillText('Client', PAD + 28, 212);
  ctx.fillStyle = '#111';
  ctx.font = font(44, 700);
  ctx.fillText(client.code, PAD + 28, 264);
  const cur = allWeeks[allWeeks.length - 1];
  ctx.textAlign = 'right';
  ctx.fillStyle = '#555';
  ctx.font = font(24);
  ctx.fillText('Reste à payer', W - PAD - 28, 212);
  ctx.fillStyle = cur.remaining > 0 ? '#c0392b' : '#1e8449';
  ctx.font = font(46, 700);
  ctx.fillText(`${fmt(cur.remaining)} F`, W - PAD - 28, 266);
  let y = HEADER;
  lines.forEach((l) => {
    if (l.kind === 'week') {
      y += 26 + LINE;
      ctx.textAlign = 'left';
      ctx.fillStyle = '#111';
      ctx.font = font(32, 700);
      ctx.fillText(l.left, PAD, y - 14);
      ctx.textAlign = 'right';
      ctx.fillStyle = '#777';
      ctx.font = font(24);
      ctx.fillText(l.right, W - PAD, y - 14);
      ctx.fillStyle = '#111';
      ctx.fillRect(PAD, y - 2, W - PAD * 2, 3);
    } else if (l.kind === 'row') {
      y += LINE;
      ctx.textAlign = 'left';
      ctx.fillStyle = '#777';
      ctx.font = font(24);
      if (l.date) ctx.fillText(l.date, PAD, y - 14);
      ctx.fillStyle = l.color === '#777' ? '#777' : '#111';
      ctx.font = font(26);
      ctx.fillText(stmtTruncate(ctx, l.left, 600), PAD + 150, y - 14);
      ctx.textAlign = 'right';
      ctx.fillStyle = l.color || '#111';
      ctx.font = font(28, 600);
      ctx.fillText(l.right, W - PAD, y - 14);
      ctx.fillStyle = '#eeeeee';
      ctx.fillRect(PAD, y - 1, W - PAD * 2, 2);
    } else {
      y += LINE + 10;
      ctx.textAlign = 'left';
      ctx.fillStyle = '#111';
      ctx.font = font(26, 700);
      ctx.fillText(l.left, PAD, y - 14);
      ctx.textAlign = 'right';
      ctx.font = font(30, 700);
      ctx.fillText(l.right, W - PAD, y - 14);
    }
  });
  if (weeks.length < allWeeks.length) {
    ctx.textAlign = 'center';
    ctx.fillStyle = '#999';
    ctx.font = font(22);
    ctx.fillText('Semaines plus anciennes : voir la version imprimée', W / 2, canvas.height - 50);
  }
  return canvas;
}

function useCountUp(target, duration = 500) {
 const [display, setDisplay] = useState(target);
 const prevRef = useRef(target);
 useEffect(() => {
 const from = prevRef.current;
 const to = target;
 if (from === to) return;
 const start = performance.now();
 let frame;
 const tick = (now) => {
 const elapsed = now - start;
 const progress = Math.min(1, elapsed / duration);
 const eased = 1 - Math.pow(1 - progress, 3);
 setDisplay(from + (to - from) * eased);
 if (progress < 1) frame = requestAnimationFrame(tick);
 else prevRef.current = to;
 };
 frame = requestAnimationFrame(tick);
 return () => cancelAnimationFrame(frame);
 }, [target, duration]);
 return display;
}

function QuickCalc() {
 const [weight, setWeight] = useState('');
 const [rate, setRate] = useState('');
 const [activeField, setActiveField] = useState('weight');
 const w = parseFloat(weight.replace(',','.'));
 const r = parseFloat(rate.replace(',','.'));
 const hasResult = weight && rate && !isNaN(w) && !isNaN(r);

 const press = (key) => {
 const setter = activeField === 'weight' ? setWeight : setRate;
 setter((v) => {
 if (key === 'back') return v.slice(0, -1);
 if (key === '.') return v.includes('.') ? v : v + (v === '' ? '0.' : '.');
 return v + key;
 });
 };

 const clear = () => {
 setWeight('');
 setRate('');
 };

 return (
 <div className="rounded-2xl px-4 py-4" style={{ background: C.surface, border: `1px solid ${C.border}` }}>
 <div className="flex gap-2 mb-3">
 <AmountDisplay label="Poids (kg)" value={weight} active={activeField === 'weight'} onClick={() => setActiveField('weight')} />
 <AmountDisplay label="Tarif (par kg)" value={rate} active={activeField === 'rate'} onClick={() => setActiveField('rate')} />
 </div>
 <div className="flex items-center justify-between rounded-2xl px-4 py-3 mb-3" style={{ background: C.surface2, border: `1px solid ${C.border}` }}>
 <div style={{ color: C.textFaint, fontSize: '11.5px', fontFamily:"'Oswald', sans-serif", letterSpacing: '0.06em', textTransform: 'uppercase' }}>Résultat</div>
 <div style={{ color: hasResult ? C.amber : C.textFaint, fontFamily:"'JetBrains Mono', monospace", fontSize: '19px', fontWeight: 600 }}>
 {hasResult ? `${fmt(w * r)}` : '—'}
 </div>
 </div>
 <NumPad onPress={press} />
 <button onClick={clear} className="w-full py-2" style={{ color: C.textSecondary, fontFamily:"'Oswald', sans-serif", fontSize: '12.5px', fontWeight: 600 }}>
 Effacer
 </button>
 </div>
 );
}

function WeeklyTrendChart({ trend }) {
  const n = trend.length;
  const [active, setActive] = useState(Math.max(0, n - 1));
  useEffect(() => {
    setActive(Math.max(0, n - 1));
  }, [n]);
  const gid = useMemo(() => 'tc' + Math.random().toString(36).slice(2, 8), []);
  if (n === 0) return null;

  const W = 320;
  const H = 150;
  const PX = 16;
  const PT = 12;
  const PB = 12;
  const maxVal = Math.max(1, ...trend.map((w) => Math.max(w.paid || 0, w.due || 0)));
  const step = n > 1 ? (W - PX * 2) / (n - 1) : 0;
  const xAt = (i) => (n > 1 ? PX + i * step : W / 2);
  const yAt = (v) => PT + (1 - (v || 0) / maxVal) * (H - PT - PB);
  const smoothPath = (vals) =>
    vals
      .map((v, i) => {
        const x = xAt(i);
        const y = yAt(v);
        if (i === 0) return `M ${x} ${y}`;
        const px = xAt(i - 1);
        const py = yAt(vals[i - 1]);
        const mx = (px + x) / 2;
        return `C ${mx} ${py}, ${mx} ${y}, ${x} ${y}`;
      })
      .join(' ');
  const paidVals = trend.map((w) => w.paid || 0);
  const dueVals = trend.map((w) => w.due || 0);
  const paidPath = smoothPath(paidVals);
  const duePath = smoothPath(dueVals);
  const areaPath = `${paidPath} L ${xAt(n - 1)} ${H - PB} L ${xAt(0)} ${H - PB} Z`;

  const idx = Math.min(Math.max(active, 0), n - 1);
  const cur = trend[idx];
  const prev = idx > 0 ? trend[idx - 1] : null;
  const delta = prev && prev.paid > 0 ? ((cur.paid - prev.paid) / prev.paid) * 100 : null;
  const rate = cur.due > 0 ? Math.round((cur.paid / cur.due) * 100) : 0;

  const pick = (clientX, el) => {
    const rect = el.getBoundingClientRect();
    if (!rect.width) return;
    const vx = ((clientX - rect.left) / rect.width) * W;
    const i = n > 1 ? Math.round((vx - PX) / step) : 0;
    setActive(Math.max(0, Math.min(n - 1, i)));
  };

  return (
    <div className="chart-in rounded-3xl px-4 pt-4 pb-3 mb-6" style={{ background: C.surface, border: `1px solid ${C.border}` }}>
      <div className="flex items-start justify-between mb-1 gap-3">
        <div className="min-w-0">
          <div style={{ color: C.textFaint, fontSize: '10.5px', fontFamily: "'Oswald', sans-serif", letterSpacing: '0.06em', textTransform: 'uppercase', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
            {cur.label}
          </div>
          <div style={{ color: C.green, fontFamily: "'JetBrains Mono', monospace", fontSize: '21px', fontWeight: 700, lineHeight: 1.25 }}>{fmt(cur.paid)}</div>
          <div style={{ color: C.textSecondary, fontSize: '11px', fontFamily: "'JetBrains Mono', monospace" }}>
            sur {fmt(cur.due)} dû · {rate}% recouvré
          </div>
        </div>
        {delta !== null && (
          <div className="rounded-full px-2.5 py-1 flex-shrink-0" style={{ background: delta >= 0 ? 'rgba(74,222,128,0.12)' : 'rgba(251,113,133,0.12)', color: delta >= 0 ? C.green : C.rust, fontSize: '11.5px', fontFamily: "'JetBrains Mono', monospace", fontWeight: 700 }}>
            {delta >= 0 ? '▲' : '▼'} {Math.abs(delta).toFixed(0)}%
          </div>
        )}
      </div>

      <div
        style={{ touchAction: 'pan-y', cursor: 'pointer', userSelect: 'none', WebkitUserSelect: 'none' }}
        onPointerDown={(e) => pick(e.clientX, e.currentTarget)}
        onPointerMove={(e) => {
          if (e.pointerType === 'mouse' && e.buttons === 0) return;
          pick(e.clientX, e.currentTarget);
        }}
      >
        <svg viewBox={`0 0 ${W} ${H}`} width="100%" style={{ display: 'block', overflow: 'visible' }}>
          <defs>
            <linearGradient id={`${gid}a`} gradientUnits="userSpaceOnUse" x1="0" y1={PT} x2="0" y2={H - PB}>
              <stop offset="0%" stopColor="#ff9d42" stopOpacity="0.32" />
              <stop offset="100%" stopColor="#ff9d42" stopOpacity="0" />
            </linearGradient>
            <linearGradient id={`${gid}l`} gradientUnits="userSpaceOnUse" x1="0" y1="0" x2={W} y2="0">
              <stop offset="0%" stopColor="#ffb066" />
              <stop offset="100%" stopColor="#ff6f61" />
            </linearGradient>
          </defs>
          {[0, 0.5, 1].map((f) => (
            <line key={f} x1={PX} x2={W - PX} y1={PT + f * (H - PT - PB)} y2={PT + f * (H - PT - PB)} stroke="rgba(255,255,255,0.07)" strokeWidth="1" />
          ))}
          <path d={duePath} fill="none" stroke="rgba(255,255,255,0.3)" strokeWidth="1.5" strokeDasharray="4 5" />
          <path d={areaPath} fill={`url(#${gid}a)`} />
          <path d={paidPath} fill="none" stroke={`url(#${gid}l)`} strokeWidth="3" strokeLinecap="round" strokeLinejoin="round" />
          <line x1={xAt(idx)} x2={xAt(idx)} y1={PT - 6} y2={H - PB} stroke="rgba(255,255,255,0.22)" strokeWidth="1" />
          {paidVals.map((v, i) => (
            <circle key={i} cx={xAt(i)} cy={yAt(v)} r={i === idx ? 6 : 2.5} fill={i === idx ? C.bg : '#ff9d42'} stroke={i === idx ? '#ff9d42' : 'none'} strokeWidth={i === idx ? 3 : 0} />
          ))}
        </svg>
      </div>

      <div className="relative mt-1" style={{ height: '16px' }}>
        {trend.map((w, i) => (
          <div key={i} className="absolute" style={{ left: `${(xAt(i) / W) * 100}%`, transform: 'translateX(-50%)', color: i === idx ? C.amber : C.textFaint, fontSize: '10px', fontFamily: "'JetBrains Mono', monospace", fontWeight: i === idx ? 700 : 400 }}>
            S{i + 1}
          </div>
        ))}
      </div>
      <div className="flex items-center gap-4 mt-2" style={{ fontSize: '10.5px', color: C.textFaint }}>
        <div className="flex items-center gap-1.5">
          <span style={{ width: '14px', height: '3px', borderRadius: '2px', background: C.amberGradient, display: 'inline-block' }} /> Encaissé
        </div>
        <div className="flex items-center gap-1.5">
          <span style={{ width: '14px', height: '0', borderTop: '2px dashed rgba(255,255,255,0.35)', display: 'inline-block' }} /> Dû
        </div>
        <div style={{ marginLeft: 'auto' }}>Glisse le doigt pour explorer</div>
      </div>
    </div>
  );
}

function ClientCardInner({ c, t, selectMode, isSelected, onClick }) {
 const statusColor = { paye: C.green, partiel: C.amber, impaye: C.rust }[t.status];
 const [showQuickActions, setShowQuickActions] = useState(false);
 const [dragX, setDragX] = useState(0);
 const pressTimer = useRef(null);
 const longPressed = useRef(false);
 const dragStartX = useRef(null);
 const dragging = useRef(false);
 const dragStartY = useRef(null);
 const axisRef = useRef(null);

 const startPress = () => {
 if (selectMode || !c.phone) return;
 longPressed.current = false;
 pressTimer.current = setTimeout(() => {
 longPressed.current = true;
 setShowQuickActions(true);
 }, 480);
 };
 const endPress = () => {
 if (pressTimer.current) clearTimeout(pressTimer.current);
 };
 const handleClick = () => {
 if (longPressed.current) {
 longPressed.current = false;
 return;
 }
 onClick();
 };

 const onCardTouchStart = (e) => {
 dragStartX.current = e.touches[0].clientX;
 dragStartY.current = e.touches[0].clientY;
 axisRef.current = null;
 dragging.current = true;
 startPress();
 };
 const onCardTouchMove = (e) => {
 if (!dragging.current || dragStartX.current === null) return;
 const dx = e.touches[0].clientX - dragStartX.current;
 const dy = e.touches[0].clientY - dragStartY.current;
 if (Math.abs(dx) > 6 || Math.abs(dy) > 6) endPress();
 if (!axisRef.current) {
 if (Math.abs(dx) < 10 && Math.abs(dy) < 10) return;
 axisRef.current = Math.abs(dx) > Math.abs(dy) ? 'x' : 'y';
 }
 if (axisRef.current !== 'x' || selectMode || !c.phone) return;
 const next = Math.max(Math.min(dx, 0), -72);
 setDragX((prev) => (prev === next ? prev : next));
 };
 const onCardTouchEnd = () => {
 endPress();
 dragging.current = false;
 if (dragX <= -45 && c.phone && !selectMode) {
 longPressed.current = true;
 setShowQuickActions(true);
 }
 setDragX(0);
 dragStartX.current = null;
 };

 return (
 <>
 <div className="relative rounded-3xl overflow-hidden cv-auto">
 {!selectMode && c.phone && (
 <div
 className="absolute inset-y-0 right-0 flex items-center justify-center"
 style={{ width: '72px', background: 'rgba(110,168,254,0.18)', opacity: Math.min(1, -dragX / 45) }}
 >
 <MessageCircle size={20} color={C.steel} />
 </div>
 )}
 <button
 onClick={handleClick}
 onTouchStart={onCardTouchStart}
 onTouchEnd={onCardTouchEnd}
 onTouchMove={onCardTouchMove}
 onMouseDown={startPress}
 onMouseUp={endPress}
 onMouseLeave={endPress}
 className="glass rounded-3xl px-4 py-4 text-left flex gap-3 relative overflow-hidden w-full"
 style={{
 background: C.surface,
 border: `1px solid ${isSelected ? C.amber : C.borderLight}`,
 boxShadow: isSelected ? '0 0 0 3px rgba(255,157,66,0.12)' : 'none',
 transform: `translateX(${dragX}px)`,
 transition: dragging.current ? 'none' : 'transform 0.25s cubic-bezier(0.34,1.56,0.64,1)',
 }}
 >
 <div style={{ position: 'absolute', left: 0, top: 0, bottom: 0, width: '3px', background: statusColor, opacity: 0.85 }} />
 {selectMode && (
 <div
 className="w-5 h-5 rounded-full flex items-center justify-center flex-shrink-0 mt-0.5"
 style={{ background: isSelected ? C.amberGradient : 'transparent', border: `1.5px solid ${isSelected ? C.amber : C.borderLight}` }}
 >
 {isSelected && <Check size={13} color="#0a0c10" />}
 </div>
 )}
 {!selectMode && (
 <div
 className="w-10 h-10 rounded-2xl flex items-center justify-center flex-shrink-0"
 style={{ background: `${statusColor}22`, border: `1px solid ${statusColor}44` }}
 >
 <span style={{ color: statusColor, fontFamily: "'Oswald', sans-serif", fontWeight: 700, fontSize: '14px' }}>
 {c.code.slice(0, 2).toUpperCase()}
 </span>
 </div>
 )}
 <div className="flex-1 min-w-0">
 <div className="flex items-start justify-between mb-1.5">
 <div className="flex items-center gap-1.5">
 <div style={{ color: C.textPrimary, fontFamily: "'Oswald', sans-serif", fontSize: '17.5px', fontWeight: 700, letterSpacing: '0.02em' }}>
 {c.code}
 </div>
 {c.vehicleType ==='remorque' && <PackageOpen size={13} color={C.textFaint} />}
 {c.onRoute && <MapPin size={13} color={C.amber} />}
 {c.reminderDate && <BellRing size={12} color={C.steel} />}
 </div>
 <Stamp status={t.status} />
 </div>
 <div style={{ color: C.textSecondary, fontSize: '12.5px', fontFamily: "'JetBrains Mono', monospace", lineHeight: 1.5 }}>
 {breakdownLine(c)} = {fmt(t.totalDue)}
 </div>
 {t.totalDue > 0 && <ProgressBar paid={t.totalPaid} due={t.totalDue} />}
 <div className="flex items-center justify-between mt-2">
 <div style={{ color: C.green, fontSize: '12.5px', fontFamily: "'JetBrains Mono', monospace", fontWeight: 600 }}>
 Payé {fmt(t.totalPaid)}

 <div style={{ color: t.remaining > 0 ? C.rust : C.textFaint, fontSize: '13.5px', fontFamily: "'JetBrains Mono', monospace", fontWeight: 700 }}>
 Reste {fmt(t.remaining)}

 </div>

 <Sheet open={showQuickActions} onClose={() => setShowQuickActions(false)} title={c.code}>
 {c.phone && (
 <>
 <a
 href={`tel:${c.phone.replace(/\s+/g,'')}`}

 {t.remaining > 0 && (
 <a
 href={`https://wa.me/${c.phone.replace(/\D/g,'')}?text=${encodeURIComponent(`Bonjour ${c.code}, votre reste à payer est de ${fmt(t.remaining)} F. Merci de régulariser dès que possible.`)}`}
 target="_blank"
 rel="noopener noreferrer"

function ProgressBar({ paid, due }) {
  const pct = due > 0 ? Math.min(1, paid / due) : 0;
  const gradient = pct >= 1 ? C.greenGradient : pct > 0 ? C.amberGradient : C.rust;
  return (
    <div className="rounded-full overflow-hidden mt-2.5" style={{ height: '6px', background: 'rgba(255,255,255,0.08)' }}>
      <div className="bar-fill" style={{ height: '100%', width: '100%', transform: `scaleX(${pct})`, background: gradient }} />
    </div>
  );
}

 <div style={{ color: C.textSecondary }}>Payé <span style={{ color: C.green, fontWeight: 600 }}>{fmt(paid)}</span></div>
 <div style={{ color: C.textSecondary }}>Reste <span style={{ color: remaining > 0 ? C.rust : C.textFaint, fontWeight: 600 }}>{fmt(remaining)}</span></div>
 </div>
 {matchedClients.length > 0 && (
 <div className="flex flex-wrap gap-1.5 mt-2.5">
 {matchedClients.map((c) => {
 const t = clientTotals(c);
 return (
 <div
 key={c.id}
 className="rounded-full px-2.5 py-1"
 style={{ background: C.surface2, border: `1px solid ${C.amberDim}`, fontFamily:"'JetBrains Mono', monospace", fontSize: '11px' }}
 >
 <span style={{ color: C.amber, fontWeight: 600 }}>{c.code}</span>
 <span style={{ color: C.textFaint }}> · reste {fmt(t.remaining)}</span>
 </div>
 );
 })}
 </div>
 )}
 </button>
 );
 })}
 </div>
 </div>
 );
}

function ArchiveWeekDetail({ archive, onBack, onOpenClient, onExportJSON, onExportCSV }) {
 const [search, setSearch] = useState('');
 const [filterStatuses, setFilterStatuses] = useState([]);
 const [sortBy, setSortBy] = useState('nom');
 const [showFilter, setShowFilter] = useState(false);

 const rows = useMemo(() => {
 let list = archive.clients.map((c) => ({ c, t: clientTotals(c) }));
 if (search.trim()) {
 const q = search.trim().toUpperCase();
 list = list.filter((x) => x.c.code.includes(q));
 }
 if (filterStatuses.length > 0) {
 list = list.filter((x) => filterStatuses.includes(x.t.status));
 }
 if (sortBy === 'nom') list.sort((a, b) => a.c.code.localeCompare(b.c.code));
 else if (sortBy === 'reste') list.sort((a, b) => b.t.remaining - a.t.remaining);
 else if (sortBy === 'du') list.sort((a, b) => b.t.totalDue - a.t.totalDue);
 else if (sortBy === 'paye') list.sort((a, b) => b.t.totalPaid - a.t.totalPaid);
 return list;
 }, [archive, search, filterStatuses, sortBy]);

 const totals = useMemo(() => {
 let due = 0, paid = 0, remaining = 0;
 archive.clients.forEach((c) => {
 const t = clientTotals(c);
 due += t.totalDue;
 paid += t.totalPaid;
 remaining += t.remaining;
 });
 return { due, paid, remaining };
 }, [archive]);

 return (
 <div className="px-5">
 <div className="flex items-center gap-3 mb-4">
 <button onClick={onBack} className="w-9 h-9 rounded-full flex items-center justify-center flex-shrink-0" style={{ background: C.surface2, border: `1px solid ${C.border}` }}>
 <ChevronLeft size={18} color={C.textSecondary} />
 </button>
 <div style={{ flex: 1, minWidth: 0 }}>
 <div style={{ color: C.textPrimary, fontFamily:"'Oswald', sans-serif", fontSize: '17px', fontWeight: 600, lineHeight: 1.3 }}>
 {archive.label}
 </div>
 <div style={{ color: C.textFaint, fontSize: '11.5px', fontFamily:"'JetBrains Mono', monospace" }}>
 Clôturée le {fmtDate(archive.archivedAt)}
 </div>
 </div>
 </div>

 <div className="flex gap-2 mb-4">
 <StatCard label="Dû" value={totals.due} color={C.textPrimary} />
 <StatCard label="Payé" value={totals.paid} color={C.green} />
 <StatCard label="Reste" value={totals.remaining} color={totals.remaining > 0 ? C.rust : C.textFaint} />
 </div>

 <div className="flex gap-2 mb-4">
 <button
 onClick={onExportJSON}
 className="flex-1 rounded-2xl px-3 py-2.5 flex items-center justify-center gap-2"
 style={{ background: C.surface2, border: `1px solid ${C.border}` }}
 >
 <Download size={14} color={C.amber} />
 <span style={{ color: C.textPrimary, fontFamily:"'Oswald', sans-serif", fontSize: '12.5px', fontWeight: 600 }}>JSON</span>
 </button>
 <button
 onClick={onExportCSV}
 className="flex-1 rounded-2xl px-3 py-2.5 flex items-center justify-center gap-2"
 style={{ background: C.surface2, border: `1px solid ${C.border}` }}
 >
 <Download size={14} color={C.steel} />
 <span style={{ color: C.textPrimary, fontFamily:"'Oswald', sans-serif", fontSize: '12.5px', fontWeight: 600 }}>CSV</span>
 </button>
 </div>

 <div className="flex gap-2 mb-4">
 <div className="flex-1 flex items-center gap-2 rounded-2xl px-3 py-2.5" style={{ background: C.surface2, border: `1px solid ${C.border}` }}>
 <Search size={16} color={C.textFaint} />
 <input
 value={search}
 onChange={(e) => setSearch(e.target.value)}
 placeholder="Rechercher un client…"
 className="w-full bg-transparent outline-none"
 style={{ color: C.textPrimary, fontFamily:"'JetBrains Mono', monospace", fontSize: '14px' }}
 />
 </div>
 <button
 onClick={() => setShowFilter(true)}
 className="w-11 h-11 rounded-2xl flex items-center justify-center"
 style={{ background: C.surface2, border: `1px solid ${C.border}` }}
 >
 <Filter size={17} color={filterStatuses.length > 0 || sortBy !== 'nom' ? C.amber : C.textFaint} />
 </button>
 </div>

 {rows.length === 0 && (
 <div className="text-center py-12" style={{ color: C.textFaint, fontSize: '13px' }}>
 Aucun client ne correspond à cette recherche.
 </div>
 )}

 <div className="flex flex-col gap-2.5 mb-6">
 {rows.map(({ c, t }) => (
 <ClientCard key={c.id} c={c} t={t} selectMode={false} isSelected={false} onClick={() => onOpenClient(c.id)} />
 ))}
 </div>

 <Sheet open={showFilter} onClose={() => setShowFilter(false)} title="Filtrer et trier">
 <div style={{ color: C.textSecondary, fontSize: '12px', fontFamily:"'Oswald', sans-serif", letterSpacing: '0.06em', textTransform: 'uppercase', marginBottom: '10px' }}>
 Statut (plusieurs choix possibles)
 </div>
 <div className="mb-5">
 <StatusChips
 selected={filterStatuses}
 onClear={() => setFilterStatuses([])}
 onToggle={(key) =>
 setFilterStatuses((prev) => (prev.includes(key) ? prev.filter((k) => k !== key) : [...prev, key]))
 }
 />
 </div>
 <div style={{ color: C.textSecondary, fontSize: '12px', fontFamily:"'Oswald', sans-serif", letterSpacing: '0.06em', textTransform: 'uppercase', marginBottom: '10px' }}>
 Trier par
 </div>
 <div className="flex gap-2 flex-wrap">
 {[
 ['nom','Nom'],
 ['reste','Reste à payer'],
 ['du','Total dû'],
 ['paye','Total payé'],
 ].map(([key, label]) => (
 <button
 key={key}
 onClick={() => setSortBy(key)}
 className="px-4 py-2 rounded-full"
 style={{
 background: sortBy === key ? C.steel : C.surface2,
 color: sortBy === key ? '#0a0c10' : C.textSecondary,
 fontFamily:"'Oswald', sans-serif",
 fontSize: '13px',
 fontWeight: 500,
 border: `1px solid ${sortBy === key ? C.steel : C.border}`,
 }}
 >
 {label}
 </button>
 ))}
 </div>
 <BigButton onClick={() => setShowFilter(false)}>Appliquer</BigButton>
 </Sheet>
 </div>
 );
}
